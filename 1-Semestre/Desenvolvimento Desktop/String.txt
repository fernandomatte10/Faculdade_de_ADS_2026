#include <stdio.h>
#include <string.h>

int main() {
    
    char senha[50], valsenha[50];
    char email[50], valemail[50];
    char nome[50];
    int idade, menu, tamanhonome;
    
    printf("Digite seu nome completo: \n");
    fgets(nome, sizeof(nome), stdin);
    nome[strcspn(nome, "\n")] = '\0';
    
    printf("Digite um email valido: \n");
    fgets(email, sizeof(email), stdin);
    email[strcspn(email, "\n")] = '\0';
    
    printf("Digite uma senha: \n");
    fgets(senha, sizeof(senha), stdin);
    senha[strcspn(senha, "\n")] = '\0';
    
    printf("Digite sua idade: \n");
    scanf("%d", &idade);
    printf("Conta criada com sucesso\n\n");
    
    tamanhonome = strlen(nome);
    
    while(1){
        printf("Menu\n\n1 - Mostrar dados do usuario\n2 - Analisar nome\n3 - Validar email\n4 - Fazer login\n5 - Sair\n\n");
        scanf("%d", &menu);
        getchar();
    
        switch(menu){
            case 1:
                printf("Dados do usuario: \n\n");
                printf("Nome: %s\nEmail: %s\nIdade: %d\nSituacao da idade: %s\n\n", nome, email, idade, idade >= 18 ? "Maior de idade" : "Menor de idade");
                break;
            case 2:
                printf("Seu nome tem %d caracteres.\n\n", tamanhonome);
                break;
            case 3:
                if (strchr(email, '@') != NULL && strstr(email, ".com") != NULL) {
                    printf("O email e valido.\n\n");
                } else {
                    printf("O email e invalido.\n\n");
                }    
                break;
            case 4:
                printf("Digite seu email cadastrado: \n");
                fgets(valemail, sizeof(valemail), stdin);
                valemail[strcspn(valemail, "\n")] = '\0';
                
                if (strcmp(valemail, email) == 0) {
                    printf("O email e valido.\n\n");
                } else {
                    printf("Email incorreto\n");
                }
                
                printf("Digite sua senha: \n");
                fgets(valsenha, sizeof(valsenha), stdin);
                valsenha[strcspn(valsenha, "\n")] = '\0';
                
                if (strcmp(valsenha, senha) == 0) {
                    printf("Senha correta.\n");
                } else {
                    printf("Senha incorreta\n");
                }
                break;
            case 5:
                printf("Saindo do sistema...\n");
                return 0;
            default:
                printf("Opcao invalida\n\n");
                break;
        }
    }    
    
    return 0;
}
