#include <stdio.h>

int main() {
    
    int opcao;
	double base, expoente, resultado;
	
	printf("1 - Metros para centimetros \n2 - Metros para milimetros \n3 - Metros para quilometros\n");
	
	printf("Escolha uma opcao:\n");
	scanf("%d", &opcao);
	
	switch(opcao) {
	    
	    case 1:
	    printf("Digite o numero: ");
	    scanf("%lf", &base);
	    
	    resultado = base * 100;
	    printf("%.2lf metro = %.2lf centimetros \n", base, resultado);
	    break;
	    
	    case 2:
	    printf("Digite o numero: ");
	    scanf("%lf", &base);
	    
	    resultado = base * 1000;
	    printf("%.2lf metros = %.2lf milimetros\n", base, resultado);
	    break;
	    
	    case 3:
	    printf("Digite o numero: ");
	    scanf("%lf", &base);
	    
	    resultado = base / 1000;
	    printf("%.2lf metros = %.4lf quilometros\n", base, resultado);
	    break;
	    
	    default:
	    printf("Opcao invalida\n");
	    return 0;
	    
	}
	
	return 0;
}
